cmake_minimum_required(VERSION 3.20)
project(rayvox_dlss LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Rayvox DLSS requires Windows x64")
endif()
find_package(JNI REQUIRED)
file(TO_CMAKE_PATH "${DLSS_SDK}" DLSS_SDK)
file(TO_CMAKE_PATH "${VULKAN_HEADERS}" VULKAN_HEADERS)
find_path(VULKAN_INCLUDE_DIR vulkan/vulkan.h HINTS "${VULKAN_HEADERS}/include" "$ENV{VULKAN_SDK}/Include" REQUIRED)
find_path(NGX_INCLUDE_DIR nvsdk_ngx_vk.h HINTS "${DLSS_SDK}/include" REQUIRED)
find_library(NGX_STATIC_LIBRARY nvsdk_ngx_s HINTS "${DLSS_SDK}/lib/Windows_x86_64/x64" REQUIRED)
add_library(rayvox_dlss SHARED dlss.cpp)
target_compile_features(rayvox_dlss PRIVATE cxx_std_17)
target_include_directories(rayvox_dlss PRIVATE ${JNI_INCLUDE_DIRS} ${VULKAN_INCLUDE_DIR} ${NGX_INCLUDE_DIR})
target_compile_definitions(rayvox_dlss PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX VK_NO_PROTOTYPES)
target_link_libraries(rayvox_dlss PRIVATE ${NGX_STATIC_LIBRARY} version)
set_property(TARGET rayvox_dlss PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
install(TARGETS rayvox_dlss RUNTIME DESTINATION .)
install(FILES "${DLSS_SDK}/lib/Windows_x86_64/rel/nvngx_dlss.dll" DESTINATION .)
install(FILES "${DLSS_SDK}/LICENSE.txt" DESTINATION . RENAME NVIDIA-DLSS-LICENSE.txt)
